{{- /* Markdown images go through Hugo: resized, converted to webp, and emitted with explicit width and height so nothing shifts on load. A remote or missing source falls back to a plain img rather than failing the build. */ -}} {{- $src := .Destination -}} {{- $img := or (.PageInner.Resources.GetMatch $src) (resources.Get $src) -}} {{- if $img }} {{- $wide := 1400 -}} {{- $narrow := 700 -}} {{- $base := cond (gt $img.Width $narrow) ($img.Resize (printf "%dx webp q82" $narrow)) $img -}} {{- $set := slice (printf "%s %dw" $base.RelPermalink $base.Width) -}} {{- if gt $img.Width $wide }} {{- $x2 := $img.Resize (printf "%dx webp q82" $wide) -}} {{- $set = $set | append (printf "%s %dw" $x2.RelPermalink $x2.Width) -}} {{- end }} {{- else }} {{- end -}}
